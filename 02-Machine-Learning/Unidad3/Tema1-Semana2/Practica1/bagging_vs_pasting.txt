# ==========================================================
# BAGGING vs PASTING
# Adult Census Income Dataset
# ==========================================================

# ==========================================================
# 1. IMPORT LIBRARIES
# ==========================================================

import pandas as pd
import numpy as np

from sklearn.datasets import fetch_openml

from sklearn.model_selection import train_test_split

from sklearn.preprocessing import (
    OneHotEncoder,
    StandardScaler
)

from sklearn.compose import ColumnTransformer

from sklearn.tree import DecisionTreeClassifier

from sklearn.ensemble import BaggingClassifier

from sklearn.metrics import (
    accuracy_score,
    classification_report,
    confusion_matrix
)

# ==========================================================
# 2. LOAD DATASET
# ==========================================================

print("\nLoading dataset...\n")

adult = fetch_openml(
    name='adult',
    version=2,
    as_frame=True
)

# Features and target
X = adult.data
y = adult.target

# ==========================================================
# 3. BASIC EXPLORATION
# ==========================================================

print("================================================")
print("DATASET INFORMATION")
print("================================================")

print("\nDataset Shape:")
print(X.shape)

print("\nFirst 5 rows:")
print(X.head())

print("\nTarget Distribution:")
print(y.value_counts())

# ==========================================================
# 4. IDENTIFY COLUMN TYPES
# ==========================================================

num_cols = X.select_dtypes(
    include=['int64', 'float64']
).columns

cat_cols = X.select_dtypes(
    include=['object', 'category']
).columns

print("\n================================================")
print("COLUMN TYPES")
print("================================================")

print("\nNumerical Columns:")
print(list(num_cols))

print("\nCategorical Columns:")
print(list(cat_cols))

# ==========================================================
# 5. TRAIN / TEST SPLIT
# ==========================================================

X_train, X_test, y_train, y_test = train_test_split(

    X,
    y,

    test_size=0.25,

    random_state=42,

    stratify=y
)

print("\n================================================")
print("TRAIN / TEST SPLIT")
print("================================================")

print("Training samples:", X_train.shape[0])
print("Testing samples :", X_test.shape[0])

# ==========================================================
# 6. PREPROCESS NUMERICAL VARIABLES
# ==========================================================

scaler = StandardScaler()

# Fit scaler ONLY on training data
X_train_num = scaler.fit_transform(
    X_train[num_cols]
)

# Transform test data
X_test_num = scaler.transform(
    X_test[num_cols]
)

# ==========================================================
# 7. PREPROCESS CATEGORICAL VARIABLES
# ==========================================================

encoder = OneHotEncoder(
    handle_unknown='ignore'
)

# Fit encoder ONLY on training data
X_train_cat = encoder.fit_transform(
    X_train[cat_cols]
)

# Transform test data
X_test_cat = encoder.transform(
    X_test[cat_cols]
)

# ==========================================================
# 8. COMBINE NUMERICAL + CATEGORICAL VARIABLES
# ==========================================================

# Convert numerical arrays to DataFrame
X_train_num_df = pd.DataFrame(
    X_train_num,
    columns=num_cols,
    index=X_train.index
)

X_test_num_df = pd.DataFrame(
    X_test_num,
    columns=num_cols,
    index=X_test.index
)

# Convert categorical sparse matrix to DataFrame
X_train_cat_df = pd.DataFrame.sparse.from_spmatrix(
    X_train_cat,
    columns=encoder.get_feature_names_out(cat_cols),
    index=X_train.index
)

X_test_cat_df = pd.DataFrame.sparse.from_spmatrix(
    X_test_cat,
    columns=encoder.get_feature_names_out(cat_cols),
    index=X_test.index
)

# Merge numerical and categorical features
X_train_processed = pd.concat(
    [X_train_num_df, X_train_cat_df],
    axis=1
)

X_test_processed = pd.concat(
    [X_test_num_df, X_test_cat_df],
    axis=1
)

print("\n================================================")
print("PREPROCESSED DATA")
print("================================================")

print("\nProcessed training shape:")
print(X_train_processed.shape)

print("\nProcessed testing shape:")
print(X_test_processed.shape)

# ==========================================================
# 9. BAGGING MODEL
# ==========================================================
# bootstrap=True
# Sampling WITH replacement
# ==========================================================

print("\n================================================")
print("TRAINING BAGGING MODEL")
print("================================================")

bagging_model = BaggingClassifier(

    estimator=DecisionTreeClassifier(

        max_depth=10,

        random_state=42
    ),

    n_estimators=50,

    max_samples=0.8,

    bootstrap=True,   # BAGGING

    random_state=42,

    n_jobs=-1
)

# Train model
bagging_model.fit(
    X_train_processed,
    y_train
)

# Predictions
y_pred_bag = bagging_model.predict(
    X_test_processed
)

# ==========================================================
# 10. BAGGING EVALUATION
# ==========================================================

bag_accuracy = accuracy_score(
    y_test,
    y_pred_bag
)

print("\n================================================")
print("BAGGING RESULTS")
print("================================================")

print(f"\nAccuracy: {bag_accuracy:.4f}")

print("\nConfusion Matrix:")
print(confusion_matrix(y_test, y_pred_bag))

print("\nClassification Report:")
print(classification_report(y_test, y_pred_bag))

# ==========================================================
# 11. PASTING MODEL
# ==========================================================
# bootstrap=False
# Sampling WITHOUT replacement
# ==========================================================

print("\n================================================")
print("TRAINING PASTING MODEL")
print("================================================")

pasting_model = BaggingClassifier(

    estimator=DecisionTreeClassifier(

        max_depth=10,

        random_state=42
    ),

    n_estimators=50,

    max_samples=0.8,

    bootstrap=False,  # PASTING

    random_state=42,

    n_jobs=-1
)

# Train model
pasting_model.fit(
    X_train_processed,
    y_train
)

# Predictions
y_pred_paste = pasting_model.predict(
    X_test_processed
)

# ==========================================================
# 12. PASTING EVALUATION
# ==========================================================

paste_accuracy = accuracy_score(
    y_test,
    y_pred_paste
)

print("\n================================================")
print("PASTING RESULTS")
print("================================================")

print(f"\nAccuracy: {paste_accuracy:.4f}")

print("\nConfusion Matrix:")
print(confusion_matrix(y_test, y_pred_paste))

print("\nClassification Report:")
print(classification_report(y_test, y_pred_paste))

# ==========================================================
# 13. FINAL COMPARISON
# ==========================================================

print("\n================================================")
print("FINAL COMPARISON")
print("================================================")

print(f"\nBagging Accuracy : {bag_accuracy:.4f}")

print(f"Pasting Accuracy : {paste_accuracy:.4f}")

# END
